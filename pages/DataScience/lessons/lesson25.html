# BƯỚC 5: CODE HOÀN CHỈNH (BẠN CẦN CHẠY)

import pandas as pd
import statsmodels.formula.api as smf
import matplotlib.pyplot as plt

# 1. Đọc dữ liệu
df = pd.read_csv("data.csv")

# 2. Mô hình tốt nhất: chỉ Duration
model = smf.ols('Calorie_Burnage ~ Duration', data=df).fit()

# 3. In kết quả
print(model.summary())

# 4. Dự đoán
def predict_calo(duration):
    return model.params['Intercept'] + model.params['Duration'] * duration

print(f"60 phút tập → {predict_calo(60):.0f} calo")
print(f"90 phút tập → {predict_calo(90):.0f} calo")

# 5. Vẽ đồ thị
plt.scatter(df['Duration'], df['Calorie_Burnage'], alpha=0.6)
plt.plot(df['Duration'], model.predict(), color='red', linewidth=2)
plt.xlabel('Thời gian tập (phút)')
plt.ylabel('Calo đốt cháy')
plt.title(f'R² = {model.rsquared:.3f} | Mỗi phút +{model.params["Duration"]:.1f} calo')
plt.grid(True, alpha=0.3)
plt.show()

60 phút tập → 700 calo
90 phút tập → 1050 calo
