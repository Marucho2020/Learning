Lesson 17 -- React Form Select -//



# 1. Select trong React là gì?
	Trong HTML, giá trị được chọn trong <select> được xác định bằng thuộc tính selected trên thẻ <option>. Trong React, <select> là một controlled component, sử dụng thuộc tính value trên thẻ <select> và sự kiện onChange để quản lý giá trị được chọn thông qua state.


# 2. Phân tích ví dụ từ W3Schools

import { useState } from 'react';
import { createRoot } from 'react-dom/client';

function MyForm() {
  const [myCar, setMyCar] = useState("Volvo");

  const handleChange = (event) => {
    setMyCar(event.target.value);
  };

  return (
    <form>
      <select value={myCar} onChange={handleChange}>
        <option value="Ford">Ford</option>
        <option value="Volvo">Volvo</option>
        <option value="Fiat">Fiat</option>
      </select>
    </form>
  );
}

createRoot(document.getElementById('root')).render(<MyForm />);



# 3. So sánh với <input> và <textarea>

	- <select> hoạt động tương tự <input> và <textarea>: đều là controlled components sử dụng value và onChange để quản lý state.
	- Khác biệt: <select> quản lý danh sách các tùy chọn (<option>), giá trị được chọn là value của <option>.


# 4. Kết hợp với React Submit Forms 

function handleSubmit(e) {

# 	Kết hợp với Textarea và Input: 
	
	Tạo biểu mẫu kết hợp <input>, <textarea>, và <select>:
	
function MyForm() {
  const [formData, setFormData] = useState({
    name: '',
    comment: '',
    car: 'Volvo',
  });

  const handleChange = (e) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    alert(`Name: ${formData.name}, Comment: ${formData.comment}, Car: ${formData.car}`);
  };

  return (
    <form onSubmit={handleSubmit}>
      <input type="text" name="name" value={formData.name} onChange={handleChange} />
      <textarea name="comment" value={formData.comment} onChange={handleChange} />
      <select name="car" value={formData.car} onChange={handleChange}>
        <option value="Ford">Ford</option>
        <option value="Volvo">Volvo</option>
        <option value="Fiat">Fiat</option>
      </select>
      <input type="submit" value="Submit" />
    </form>
  );
}	
	
	
# 	Kết hợp với React Transitions 
	
import { CSSTransition } from 'react-transition-group';
import './style.css';

function MyForm() {
  const [myCar, setMyCar] = useState("Volvo");
  const [submitted, setSubmitted] = useState(false);

  const handleChange = (e) => setMyCar(e.target.value);

  const handleSubmit = (e) => {
    e.preventDefault();
    setSubmitted(true);
    setTimeout(() => setSubmitted(false), 2000);
  };

  return (
    <div>
      <form onSubmit={handleSubmit}>
        <select value={myCar} onChange={handleChange}>
          <option value="Ford">Ford</option>
          <option value="Volvo">Volvo</option>
          <option value="Fiat">Fiat</option>
        </select>
        <input type="submit" value="Submit" />
      </form>
      <CSSTransition in={submitted} timeout={300} classNames="fade" unmountOnExit>
        <p className="success">Selected: {myCar}</p>
      </CSSTransition>
    </div>
  );
}
