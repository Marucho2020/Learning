Lesson 16 -- Text Area -//


#1. Textarea trong React là gì?

	Trong HTML, giá trị của <textarea> được đặt giữa thẻ mở và đóng (<textarea>Giá trị</textarea>). Trong React, <textarea> hoạt động giống <input>: giá trị được quản lý qua thuộc tính value và sự kiện onChange, biến nó thành controlled component (thành phần được kiểm soát).


# 2. Phân tích ví dụ


import { useState } from 'react';
import { createRoot } from 'react-dom/client';

function MyForm() {
  const [mytxt, setMytxt] = useState("");

  function handleChange(e) {
    setMytxt(e.target.value);
  }

  return (
    <form>
      <label>Write here:
        <textarea
          value={mytxt}
          onChange={handleChange}
        />
      </label>
      <p>Current value: {mytxt}</p>
    </form>
  )
}

createRoot(document.getElementById('root')).render(<MyForm />);


# 3. So sánh với <input>

	- <textarea> trong React hoạt động gần giống <input>: cả hai đều dùng value và onChange để tạo controlled component.
	- Khác biệt chính: <textarea> cho phép nhập văn bản nhiều dòng, trong khi <input type="text"> chỉ cho phép một dòng.



# 4. Kết hợp với React Submit Forms: 

function handleSubmit(e) {
  e.preventDefault();
  alert(`Submitted: ${mytxt}`);
  setMytxt(""); // Reset textarea
}

return (
  <form onSubmit={handleSubmit}>
    <label>Write here:
      <textarea value={mytxt} onChange={handleChange} />
    </label>
    <input type="submit" value="Submit" />
    <p>Current value: {mytxt}</p>
  </form>
);
