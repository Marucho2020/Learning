Print trong Python 3.X là một hàm, như đã giới thiệu. Còn với Python 2.X nó là một câu lệnh.

1 . Output 
from time import sleep # nhập hàm sleep từ thư viện time
sleep(3)			# dừng 3 giây 

print(*object , sep='' , end='\n' , file=sys.stdout, flush=False)

	
	- *object    :  là packinh argument -> gom các argument thành một Tuple 
					// để khi print sẽ thuận tiện k cần phải + các giá trị lại với nhau hay ép kiểu 
										// packing  = 1,2,3,4 
										// print(packing)		(1, 2, 3, 4)
										// print(packing , 'Iam ' , 'Iron man')	//(1, 2, 3, 4) Iam  Iron man



 



	- sep(separate )	: Default = ' ' , giá trị khai báo ký tự phân tách giữa các phần tử 
										// mặc định được đặt là space 
										// print(packing , 'Iam ' , 'Iron man' , sep='--')
										
										
										
										
										
	- end= 				: Default='\n'  , khai báo ký tự được đặt cuối print , mặc định là xuống dòng \n 

										// print(packing , 'Iam ' , 'Iron man' , sep='--' , end='.')  //(1, 2, 3, 4)--Iam --Iron man.
										
										
										
										
	-	file			: mặc định nội dung sẽ ghi vào file sys.stdout , cũng nhờ vậy ta mới thấy được nội dung đã chạy trên shell . , ta có thể ghi nó vào 1 file giông method write 									
			
										with open('test.txt' , 'w') as f :
											print('printed by print function' , file=f)
			
			
			
			
			
	- flush 			: Mặc định là False , nếu là True sẽ yêu cầu bộ đệm in ra những gì có trong bộ đệm , False thì không
