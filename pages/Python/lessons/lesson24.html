				# e.g 4  Class con sẽ override lại constructor  và method của class cha , lúc này khi khởi tạo instance của class con bắt buộc phải dùng constructor của nó 
				class SieuNhan:
					suc_manh = 100

	# Các phương thức đặc biệt
		- Special method , Magic method , Dunder method : là các tên thường được nhắc tới khi nói về method đặc biệt trong python 
		
		- phương thức đặc biệt là các phương thức được quy ước sẵn , định dạng chung của chúng là có 2 dấy _ trước và sau của tên method  :   __tênmethod__ 
		
		#####
		__str__ 	: phương thức sử dụng giống với toString() trong java , ta có thể override hàm này trong class để lấy ra được thông tin cần mong muốn của class
				# e.g 1 : Sử dụng phương thức __str__ để in ra giá trị mong muốn giống toString()
				
				class SieuNhan:
					suc_manh = 50
					def __init__(self,ten ,  vu_khi , mau_sac):
						self.ten = ten
						self.vu_khi = vu_khi
						self.mau_sac = mau_sac
					def __str__(self):											# Khai báo và phải có self mới có thể tham chiếu được vào properties của class 
						return("Sieu Nhan Ten : {} ".format(self.ten))
					
				sieu_nhan = SieuNhan("Sieu Nhan Do" , "Sung" , "Do")
				
				print(sieu_nhan.__str__())							# Sieu Nhan Ten : Sieu Nhan Do 
				
				
		
		#####
		__repr__	: gần giống với phương thức str nhưng nó sẽ in ra thông tin chi tiết của đối tượng để ta có thể khởi dựng lại nó
		
				# e.g 2 sử dụng hàm __repr__ để khai báo việc in ra giá trị của lớp 
				
				class SieuNhan:
					suc_manh = 50
					def __init__(self , ten):
						self.ten = ten
				
					def __repr__(self):
						return 'Ten :{} va duoc lay ra tu repr '.format(self.ten)
				
				sieu_nhan = SieuNhan("Sieu Nhan Do")
				print(sieu_nhan.__repr__())								# Ten :Sieu Nhan Do va duoc lay ra tu repr 
				print(sieu_nhan.__repr__)								# <bound method SieuNhan.__repr__ of Ten :Sieu Nhan Do va duoc lay ra tu repr >  Chỗ này ko có () tức k phải lời gọi hàm nên nó sẽ in ra thuộc tính repr của nó 
				
		

		#####
		__len__ 	: cũng là một special method để có thể sử dụng được 
				
				# e.g 3 : Ví dụ khai báo và sử dụng hàm __len__()
				class SieuNhan:
					suc_manh = 50
					ten = "SIEU NHAN DO"
					def __len__(self):
						return len(self.ten)
				
				sieu_nhan = SieuNhan()
				print(sieu_nhan.__len__())							// 12
				
		#####
			__add__  : cũng là một special method và có thể sử dụng được 
				class SieuNhan:
					suc_manh = 50
					ten = "SIEU NHAN DO"
					def __add__(self , mot_sieu_nhan_khac):
						return self.suc_manh + mot_sieu_nhan_khac.suc_manh
				sieu_nhan = SieuNhan()
				sieu_nhan_2 = SieuNhan()
				print(sieu_nhan.__add__(sieu_nhan_2))					// 100


# Getter , Setter ,  Deleter 		
	- Sử dụng để làm việc với các thuộc tính ,  ta có thể tùy chỉnh sửa trước khi lấy ra giá trị của thuộc tính 
	- Giống với chức năng của getter và setter trong java 
	- tên của property và @property không được trùng lặp nhau, tức là trong một class chỉ có thể tồn tại một trong 2
	- lưu ý : KHÔNG ĐƯỢC SỬ DỤNG TÊN CỦA PROPERTY THÔNG THƯỜNG GIỐNG VỚI TÊN CỦA PROPERTY METHOD 
	
	##### GETTER 
	@property 										# sử dụng @property để khai báo đây là một thuộc tính chứ k phải một method thông thường  
	
	def email(self):

		#e.g 1 Sử dụng getter để tạo một thuộc tính mới là phương thức 
		class SieuNhan:
			suc_manh =  50
			ten = "Sieu Nhan"
			email = "EMAIL"
			
			@property
			def gmail(self):
				return self.ten+self.email+'@gmail.com'
		
		sieuNhan = SieuNhan()
		print(sieuNhan.gmail)						# Sieu NhanEMAIL@gmail.com
		
		
		
		
		##### SETTER
		@<propertyName>.setter
		
				-> @getGmail.setter
		
					#e.g 2 : Sử dụng setter để cập nhật lại giá trị khác cho thuộc tính của instance
						class SieuNhan:
							suc_manh = 100
							ten = "Sieu Nhan Do"
							email = "E-Mail"
						
							@property
							def getGmail(self):										   # đây là property getter gmail
								return self.ten +'.' +self.email+"@gmail.com"
						
							@getGmail.setter											# tại đây là property cập nhật lại gmail
							def setGmail(self , newEmail):
								self.email = newEmail
							
						sieuNhan = SieuNhan()
						print(sieuNhan.getGmail)											# Sieu Nhan Do.E-Mail@gmail.com
						sieuNhan.setGmail = "NEW-EMAIL"										# tại đây thực hiện việc set lại email mới cho property email của instance sieuNhan 
						print(sieuNhan.getGmail)											# kiểm tra lại Sieu Nhan Do.NEW-EMAIL@gmail.com
		
		
		#### Deleter
		 - là công cụ giúp xóa thuộc tính vừa gán. xóa ở đây là việc gán một GIÁ TRỊ RÁC (giá trị không có ý nghĩa ) cho cái chúng ta muốn xóa , thường là None , 0 , list rỗng tùy từng kiểu dữ liệu mà ta muốn xóa 
		 
		 @<propertyName>.deleter 
					-> @getGmail.deleter 
						def deleteGmail
				del sieuNhan.deleteGmail  		# sử dụng keyword del để báo python răng đây là thực hiện xóa 
				
		
		
				# e.g 3 sử dụng deleter để xóa tên 
				class SieuNhan:
					name = "Sieu Nhan Do"
				
					@property
					def getName(self):
						return self.name
				
					@getName.setter
					def setName(self , newName):
						self.name = newName
				
					@getName.deleter						# sử dụng @getName.deleter để khai báo property xóa 
					def deleteName(self):
						self.name = "NO-NAME"
				
				
				sieuNhan = SieuNhan()
				print(sieuNhan.getName)
				sieuNhan.setName = "SIEU NHAN XANH "
				print(sieuNhan.name)
				del sieuNhan.deleteName						# thực hiện việc xóa 
				print(sieuNhan.name)						# NO-NAME
