Lesson 11 -- Kiểu dữ liệu Set -//


	Set là một kiểu container , tuy không được sử dụng nhiều bằng LIST hay TUPLE 
		
1.	Định nghĩa 
		- Một set được giới hạn bởi cặp ngoặc {} , tất cả các phần tử cách nhau bằng dấu , 
		- SET KHÔNG CHỨA NHIỀU HƠN 1 PHẦN TỬ TRÙNG LẶP (trùng id bộ nhớ ấy ) 
		- Set CHỈ CÓ THỂ  chứa hashable object nhưung chính nó KHÔNG phải là một hashable object -> Do đó không thể có SET lồng nhau được 
		
		
		_set1 = {69,96}
		_set2 = {69,'101'}
		_tup1 = (1,)
		_set3 = {_tup1,1,"x"}
	
		print(_set1)                          // 	{96, 69}
		print(_set2)                          // 	{69, '101'}
		print(_set3)                          // 	{'x', (1,), 1}
		
		_set4 = {[]}						 //		ERROR
		print(_set4)                          // 
	
2.	Các cách khởi tạo 
	- Khởi tạo trực tiếp 
				_set3 = {_tup1,1,"x"}		
				
	- Sử dụng Set Comprehension 
				_setC = {value for value in range(3)}		// {0, 1, 2}	
				
	- Sử dụng constructor Set   // Truyền iterable thôi , truyền kiểu dữ liệu nguyên thủy là lỗi đó 
			set(iterable)	// Truyền vào các iterable , String . Tuple 
				_setC = set([1,2,3])			// {1, 2, 3}
				_empty_set = set()		
				
				_setC = set("Str")			  // {'S', 'r', 't'}
				
				
				
				
3.	Một số toán tử trong Set 
				
									_set1 = set([1,2,3,4,5])
									_set2 = set([4,5,6,7,8])
	
	-  value in <Set>			// Kiểm tra xem value có xuất hiện trong Set hay không , True or False 
	
	-  Toán tử [ - ] 			// Trả về một Set Mới  gồm các phần tử chỉ tồn tại trong Set1 mà không tồn tại trong Set2 
								// Tức là nó chỉ trừ , Bỏ đi các phần tử trùng lặp trong Set bị trừ
										print(_set1-_set2)		// {1, 2, 3}
										

	-  Toán tử [ & ] 			// Trả về một Set mới gồm các phần tử ĐỀU CÙNG TỒN TẠI trong Set 1 và Set 2 
										print(_set1&_set2)			// {4, 5}


	-  Toán tử [ | ]		// Trả về một set mới chứa TẤT CẢ CÁC PHẦN TỬ của 2 set  	, và KHÔNG có DUPLICATE 
								print(_set1|_set2)		// 	{1, 2, 3, 4, 5, 6, 7, 8}
				
				
	-  Toán tử [ ^ ]			// Trả về một set mới chứa các phần tử KHÔNG BỊ TRÙNG trong 2 set 
								// Loại bỏ các phần tử giống nhau giữa 2 Set 
	
							print(_set1^_set2)		// 		{1, 2, 3, 6, 7, 8}

6. Các phương thức của Set  
_set1 = set([1,2,3,4,5])
_set2 = set([4,5,6,7,8])
_list1 = ['a','b']
_list2 = ['b','c']
_list3 = [3,5]


# Các Phương thức tiện ích 
		
	- .clear() 			// loại bỏ các phần tử trong set  
	
	- .copy() 			// trả về một bản sao của Set , 
	
	
# Các Phương thức đối sánh 
		
		
	- .union(iterable_1 , iterable2, ....)			//phương thức union 	=   Toán tử [ | ]	
													// Trả về một set mới chứa tất cả các phần tử của set 1 và ste 2 (không trùng lặp lại cac phần tử giống nhau )
									print(_set1.union(_list1))	// {1, 2, 3, 4, 5, 'b', 'a'}	
									
									
									
	- .intersection(iterable_1,iterable_2,....)		// Phương thức iterasection = 		 Toán tử [ & ] 		
												// Trả về một set mới là Set có các giá trị cùng tồn tại trong set và iterable truyền vào 
												print(_set1.intersection(_list1, _list2,_list3))	// set()  Rỗng vì list 1 và list 2 không có các phần tử ở trong list 3 và set 
												
	- .symmetric_difference(iterable)			// 	symmetric_difference		Toán tử [ ^ ]		
												// Trả về một set mới chứa các phần tử KHÔNG tồn tại trùng nhau trong 2 set 
												// ngược lại với iterasection nhưng chỉ có 1 args 
												
												
									
									
	
# Các Phương thức cập nhật  
	
	- 	.pop() 					// Lấy ra phần tử Bé đến Lớn trong set , việc pop gia trị liên quan đến các giá trị của hàm hash trong các phần tử 
								// Đó là lý do tại sao trong set CHỈ TỒN TẠI DỮ LIỆU HASH OBJECT 
								// NẾU SET RỖNG SẼ LỖI 
						
	- 	.remove(value)			// Loại bỏ giá trị value trong set , 
								// nếu như value không trong set sẽ lỗi 

	
	
	-	.discard(value)			// loại bỏ giá trị value ở trong 
								// NẾU KHÔNG TỒN TẠI THÌ CŨNG K BÁO LỖI 
								
									
	-	.add(value)				// Thêm một phần tử vào set 
								// NẾU ĐÃ TỒN TẠI TRONG SET RỒI THÌ SẼ BỎ QUA 
								
								
	-	.update(iterable_1 , iterable_2, ....)	// Lần lượt thêm các phần tử iterable vào trong set 
												// Nó sẽ bỏ cặp ngoặc ở các iterable, cho nên nếu là list 1 cấp thì sẽ không lỗi 
												// nếu là list lồng nhau thì sẽ lỗi , còn các hash khác thì ok 
												
												
												
# Các phương thức xác thực

									// MỘT SET TRỐNG LUÔN LUÔN LÀ CON CỦA MỘT SET KHÁC 
												
	
	
	-	.issuperset(Other Set )		// return True or False
									// Kiểm tra xem  Set có phải là tập cha của OtherSet hay không
