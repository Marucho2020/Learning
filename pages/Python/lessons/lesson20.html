	// kiểu dữ liệu range(dãy số )
	
	# Khái niệm 
		- range có hỗ trợ indexing 
		- 1_000_000_000_000_000 in range(1_000_000_000_000_001)   việc khởi tạo 1 dãy số bằng range là cách tăng performance tốt nhất trong việc quản lý danh sách số 
		
	# Cách khởi tạo 1 
	range(stopNumber )   // tạo ra một dãy số bằng số 0 và kết thúc bằng stopNumber - 1 
	
	
		// e.g 1
			k = range(3)
			print(k)			// range(0, 3)
			print(k[0])			//  0
			
	
	
		
	
	# Cách khởi tạo 2 
	range(start,stop[,step])	// tạo ra một dãy số với start dến  stop -1 và step , mặc định step = 1 
								// STEP buộc phải khác 0 
	
		// e.g 2
			k = range(5 , 100,5)

			print(k[6])				// 35

	# Sự khác nhau giữa sequence scan và indexing scan 
	
		## sequence scan 
			lst = [1, 2, 3]
			for value in lst:
				value += 1
			print(lst)		// [1, 2, 3]		// KHÔNG cập nhật giá trị bên trong list 

		
		## indexing scan 
			lst = [1, 2, 3]
			for i in range(len(lst)):
				lst[i] += 1
			print(lst)		// [2, 3, 4]		// CÓ cập nhật giá trị của từng phần tử list 

		
	# Comprehension 
		- Comprehension thường được sử dụng để xử lý nhiều việc mà chỉ cần 1 dòng 
		- So sánh tốc độ của comprehension và những đoạn code với chức năng tương tự thì comprehension nhanh hơn 
		
		## Cú pháp 
		[output-expresion for-statement optional-predicate]
		
		// e.g 1 
		
		k  = ['--'.join((a.capitalize(), b.upper() + c.lower())) for a, b, c in [('how', 'kteam', 'EDUCATION'), ('chia', 'sẻ', 'FREE')]]
		print(k)	// ['How--KTEAMeducation', 'Chia--SẺfree']
		
	
		Tương đương với đoạn 
			 lst = []
			 for a, b, c in [('how', 'kteam', 'EDUCATION'), ('chia', 'sẻ', 'FREE')]:
			     a = a.capitalize()
			     b = b.upper()
			     c = c.lower()
			     lst.append('--'.join((a, b + c)))
			
			print(lst)		// ['How--KTEAMeducation', 'Chia--SẺfree']
