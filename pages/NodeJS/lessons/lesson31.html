Lesson 31 -- DNS Module -//


DNS = Domain Name System = “Danh bạ Internet”
Ví dụ: google.com → 142.250.190.78
	Trong Node.js, bạn không cần cài thêm gì, chỉ cần:

const dns = require('dns'); // hoặc require('node:dns')

→ Bạn có thể tự tra IP từ domain, tra ngược domain từ IP, kiểm tra mail server, v.v.

	Rất hay dùng khi:
		Xây dựng proxy, CDN, load balancer
		Chống spam, blacklist IP
		Logging chi tiết (biết người dùng ở quốc gia nào)
		Tự viết resolver nhanh hơn Cloudflare/Google
	


# TẤT CẢ CÁC HÀM QUAN TRỌNG NHẤT (có ví dụ chạy được ngay)

	## 1. dns.lookup() – Dùng 99% trường hợp (SIÊU HAY DÙNG) 
dns.lookup('google.com', (err, address, family) => {
  if (err) throw err;
  console.log('IP:', address);     // ví dụ: 142.250.190.78
  console.log('Family:', family);  // IPv4 hoặc IPv6
});
dns.lookup('google.com', (err, address, family) => {
  if (err) throw err;
  console.log('IP:', address);     // ví dụ: 142.250.190.78
  console.log('Family:', family);  // IPv4 hoặc IPv6
});


Kết quả:
IP: 142.250.190.78
Family: IPv4




	## Tại sao dùng `lookup` là đủ cho hầu hết app
		→ Nó dùng hệ điều hành để resolve → cực nhanh, có cache sẵn.



# 2. dns.resolve4(), dns.resolve6() – Chỉ lấy IPv4 hoặc IPv6


dns.resolve4('facebook.com', (err, addresses) => {
  console.log('IPv4 của FB:', addresses);
  // → ['157.240.241.35', ...]
});

dns.resolve6('google.com', (err, addresses) => {
  console.log('IPv6 của Google:', addresses);
});




# 3. dns.reverse() – Tra ngược: IP → Tên miền (rất hay dùng để log)

dns.reverse('8.8.8.8', (err, hostnames) => {
  if (err) {
    console.log('IP không có PTR record');
    return;
  }
  console.log('Đây là máy chủ của:', hostnames);
  // → [ 'dns.google' ]
});


# 4. dns.resolveMx() – Lấy mail server (chống spam cực mạnh)



dns.resolveMx('gmail.com', (err, addresses) => {
  if (err) throw err;
  console.log(addresses);
});
/** Kết quả:
[
  { exchange: 'gmail-smtp-in.l.google.com', priority: 5 },
  { exchange: 'alt1.gmail-smtp-in.l.google.com', priority: 10 },
  ...
]
*/

	Dùng để kiểm tra email người dùng đăng ký có thật không.


# 5. dns.resolveTxt() – Lấy TXT record (xác minh domain, SPF, DKIM)

dns.resolveTxt('google.com', (err, records) => {
  console.log(records);
  // → chứa v=spf1, google-site-verification=...
});



# 6. dns.resolveCname() – Lấy CNAME

dns.resolveCname('www.github.com', (err, addresses) => {
  console.log(addresses); // → ['github.com']
});



# 7. dns.promises – Dùng với async/await (hiện đại, sạch code)


const dns = require('dns').promises;

async function getInfo(domain) {
  try {
    const [ip, mx, txt] = await Promise.all([
      dns.lookup(domain),
      dns.resolveMx(domain),
      dns.resolveTxt(domain)
    ]);

    console.log('Domain:', domain);
    console.log('IP:', ip.address);
    console.log('Mail servers:', mx);
    console.log('TXT records:', txt.flat());
  } catch (err) {
    console.log('Không resolve được:', err.code);
  }
}

getInfo('nodejs.org');
