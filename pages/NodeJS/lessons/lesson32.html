# Assert Module là gì? Dùng để làm gì?

	assert = “khẳng định”, “đảm bảo điều này phải đúng”
	
	const assert = require('assert'); // hoặc require('node:assert')

	## 
	Mục đích duy nhất:

		Bắt lỗi ngay trong lúc phát triển khi một điều gì đó không đúng như kỳ vọng.
		→ Nếu điều kiện sai → ném lỗi và dừng chương trình ngay lập tức.


	##Dùng khi nào? 

		Viết unit test (cùng Jest, Mocha)
		Kiểm tra tham số đầu vào của hàm
		Debug logic phức tạp
		Đảm bảo contract giữa các module



# TẤT CẢ CÁC HÀM QUAN TRỌNG NHẤT

	assert(value) hoặc assert.ok(value) : value phải truthy
		assert(user);
	
	
	assert.strictEqual(a, b) 	: a === b (cùng giá trị + kiểu)
		"assert.strictEqual(status, 200)"
	
	
	
	"assert.deepStrictEqual(a, b)" : So sánh sâu object/array
			"assert.deepStrictEqual(arr1, arr2)"
			
			
	assert.throws(fn) :fn phải ném lỗi
		assert.throws(() => divide(10,0))"
		
assert.fail(message),Chủ động báo lỗi,assert.fail('Không được vào đây!')



# Ví dụ
const assert = require('assert');

// 1. Cơ bản
assert(1 + 1 === 2);        // OK
// assert(1 + 1 === 3);     // AssertionError

// 2. strictEqual (so sánh cả kiểu)
assert.strictEqual('5', 5); // AssertionError
assert.strictEqual(5, 5);   // OK

// 3. deepStrictEqual (so sánh object)
const obj1 = { name: 'Grok', age: 1 };
const obj2 = { name: 'Grok', age: 1 };
assert.deepStrictEqual(obj1, obj2); // OK

// 4. Kiểm tra hàm ném lỗi
function divide(a, b) {
  if (b === 0) throw new Error('Chia cho 0!');
  return a / b;
}

assert.throws(
  () => divide(10, 0),
  /Chia cho 0!/    // regex kiểm tra message
);

# Node.js 18+ có assert mới siêu ngon (nên dùng)

import assert from 'node:assert';

// Dùng với async/await
await assert.rejects(
  async () => { throw new Error('Lỗi rồi'); },
  /Lỗi rồi/
);

// Match lỗi chính xác
assert.match('abc123', /^[a-z]+\d+$/);
assert.doesNotMatch('abc', /^\d+$/);

# Assert vs if + throw – Nên dùng cái nào?


Trường hợp,Dùng assert,Dùng if + throw
Lỗi lập trình (developer sai),Dùng assert,Không
Lỗi người dùng (input sai),Không,Dùng throw
Trong production,assert bị tắt (NODE_ENV=production),Luôn chạy


→ assert chỉ để bắt lỗi trong lúc code, không dùng để validate user input!




# TÓM TẮT SIÊU NHANH – CHEAT SHEET



Bạn muốn kiểm tra gì?,Dùng lệnh nào?
Giá trị phải truthy,assert(value)
Hai giá trị giống nhau (cùng kiểu),"assert.strictEqual(a, b)"
Hai object/array giống nhau,"assert.deepStrictEqual(a, b)"
Hàm phải ném lỗi,assert.throws(fn)
Async function phải reject,await assert.rejects(fn)
Chủ động báo lỗi,assert.fail('Lỗi rồi')
