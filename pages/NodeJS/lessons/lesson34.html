Lesson 34 -- Readline -//

	
Readline = cách duy nhất để tương tác với người dùng trong terminal bằng Node.js
(CLI tool, hỏi mật khẩu, confirm “có chắc chắn deploy không?”, REPL…)
	
const readline = require('node:readline');  // Node.js 14+ khuyên dùng kiểu này
// hoặc const rl = require('readline');	
	
	
	
	
# 	6 CÁCH DÙNG THỰC CHIẾN (bạn sẽ dùng mãi)


	## 1. Hỏi bình thường (cơ bản nhất)

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout
});

rl.question('Bạn tên gì? ', (name) => {
  console.log(`Xin chào ${name}!`);
  rl.close();   // BẮT BUỘC close, không thì Node không thoát
});




	## 2. Dùng async/await (code sạch như React)

const rl = readline.createInterface(process.stdin, process.stdout);

const ask = (question) => new Promise(resolve => rl.question(question, resolve));

(async () => {
  const name = await ask('Tên bạn là gì? ');
  const age  = await ask('Bạn bao nhiêu tuổi? ');
  console.log(`Chào ${name}, ${age} tuổi!`);
  rl.close();
})();




	## 3. Nhập mật khẩu KHÔNG HIỆN CHỮ (login, sudo)

const askPassword = () => new Promise(resolve => {
  rl.question('Mật khẩu: ', { hideEchoBack: true }, (pass) => {
    console.log();     // xuống dòng cho đẹp
    resolve(pass);
  });
});

const pass = await askPassword();



	## 4. Confirm Yes/No siêu gọn

const confirm = async (msg = 'Bạn có chắc chắn?') => {
  const ans = await ask(`${msg} (y/N): `);
  return ['y', 'yes', 'Y', 'YES'].includes(ans.trim());
};

if (await confirm('Xóa toàn bộ database?')) {
  console.log('Đã xóa xong (đùa thôi)!');
}



	## 5. CLI đẹp như npm (có prompt, autocomplete, lịch sử)

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
  prompt: 'my-cli> ',
  completer: (line) => {
    const cmds = ['help', 'deploy', 'status', 'exit', 'clear'];
    const hits = cmds.filter(c => c.startsWith(line));
    return [hits.length ? hits : cmds, line];
  }
});

rl.prompt();

rl.on('line', (line) => {
  const cmd = line.trim();
  switch (cmd) {
    case 'help':   console.log('Các lệnh: help, deploy, exit'); break;
    case 'deploy': console.log('Đang deploy lên production...'); break;
    case 'exit':   rl.close(); break;
    default:       if (cmd) console.log(`Lệnh không tồn tại: ${cmd}`);
  }
  rl.prompt();   // hiện lại dấu >
});

rl.on('close', () => {
  console.log('Tạm biệt!');
  process.exit(0);
});


	
	
	## 6. REPL siêu nhanh (như Node REPL)

readline.createInterface({
  input: process.stdin,
  output: process.stdout,
  prompt: 'node> '
}).on('line', (code) => {
  try {
    console.log(eval(code));   // nguy hiểm, chỉ dùng dev
  } catch (e) { console.error(e); }
}).prompt();
