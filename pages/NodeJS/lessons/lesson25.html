# 1. URL Module Là Gì?	
	
	const { URL } = require('url');
	Đây là module tích hợp sẵn (built-in) của Node.js, cung cấp 2 API:
	
	
API,Mô tả,Khuyến nghị
"Legacy API (url.parse, url.format)","Cũ, từ Node.js v0.x",Không dùng cho code mới
"WHATWG URL API (new URL(), URLSearchParams)","Hiện đại, chuẩn web, mạnh mẽ",DÙNG CHO MỌI CODE MỚI	
	
	Từ Node.js 10+: URL class được ưu tiên hoàn toàn.	
	
	
	
# 2. Tại Sao Phải Dùng URL Module?	
	Vấn đề khi tự xử lý URL bằng string:	
	
const url = 'https://example.com/search?q=node.js&lang=en#section1';

// Tự cắt tay → DỄ LỖI
const query = url.split('?')[1].split('#')[0]; // 'q=node.js&lang=en'	
	
	## 	→ Rủi ro: 
		Lỗi khi URL có # trước ?
		Không xử lý Unicode, encoded chars
		Không validate protocol, domain
		Dễ bị URL injection, open redirect	
			
	
	## Giải pháp: Dùng new URL()
const myURL = new URL('https://example.com/search?q=node.js&lang=en#section1');	
	→ An toàn, chuẩn, tự động xử lý mọi thứ.
	
	
	
# 	3. Cấu Trúc Của URL Object
	
Thuộc tính,Giá trị,Mô tả
href,https://user:pass@example.com:8080/p/a/t/h?query=string#hash,Toàn bộ URL
origin,https://example.com:8080,Protocol + host
protocol,https:,Giao thức
username,user,Tên đăng nhập

searchParams,URLSearchParams,Object để thao tác query
hash,#hash,Fragment (có #)	
	
	
# 	4. URLSearchParams – "Query String Master"

const url = new URL('https://example.com/?name=Kai&age=30&city=Tokyo');
const params = url.searchParams;


Phương thức,Mô tả
get('name'),'Kai'
getAll('name'),['Kai'] (nếu có nhiều)
has('age'),true
"append('role', 'admin')",Thêm param
"set('age', '31')",Cập nhật
delete('city'),Xóa
toString(),'name=Kai&age=31&role=admin'



	## Ví dụ: Xây dựng query động
function buildSearchURL(base, filters) {
  const url = new URL(base);
  Object.entries(filters).forEach(([key, value]) => {
    if (value) url.searchParams.append(key, value);
  });
  return url.toString();
}

console.log(buildSearchURL('https://api.example.com/search', {
  q: 'node.js',
  lang: 'en',
  page: 2
}));
// → https://api.example.com/search?q=node.js&lang=en&page=2


# 5. So Sánh: Legacy API vs WHATWG API

Tính năng,Legacy (url.parse),WHATWG (new URL)
Chuẩn,Node.js cũ,Web standard (browser)
Xử lý Unicode,Không tốt,Tốt
Query parsing,String hoặc object đơn giản,URLSearchParams mạnh mẽ
Error handling,Không throw,Throw nếu URL không hợp lệ
Khuyến nghị,Không dùng,DÙNG CHO MỌI CODE MỚI



# 6. Ứng Dụng Thực Tế: File Server Dùng URL Parsing

	Tạo 2 file HTML
<h1>Summer</h1><p>I love the sun!</p>	
	
	winter.html 
<h1>Winter</h1><p>I love the snow!</p>	
	
	
	
# File server (demo_fileserver.js)
	
const http = require('http');
const url = require('url');
const fs = require('fs');
const path = require('path');

http.createServer((req, res) => {
  const parsedUrl = url.parse(req.url, true);
  const filename = "." + parsedUrl.pathname;

  // Bảo mật: Ngăn path traversal
  const safePath = path.resolve(__dirname, filename);
  if (!safePath.startsWith(__dirname)) {
    res.writeHead(400);
    return res.end('Invalid path');
  }

  fs.readFile(safePath, (err, data) => {
    if (err) {
      res.writeHead(404, { 'Content-Type': 'text/html' });
      return res.end('404 Not Found');
    }
    res.writeHead(200, { 'Content-Type': 'text/html' });
    res.write(data);
    res.end();
  });
}).listen(8080);

console.log('Server running at http://localhost:8080/');	
	
	
	
	http://localhost:8080/summer.html → Hiển thị Summer
http://localhost:8080/winter.html → Hiển thị Winter
	
	


# 7. Best Practices Khi Dùng URL Module

Best Practice,Giải thích
Luôn dùng new URL(),Thay vì url.parse()
Dùng URLSearchParams,Thao tác query an toàn
Validate URL,Dùng try-catch
Ngăn path traversal,Dùng path.resolve() + kiểm tra __dirname
Encode user input,encodeURIComponent() khi xây URL
Không expose password,Xóa password trước khi log
