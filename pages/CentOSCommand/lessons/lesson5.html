			-useradd    // tạo người dùng   ( /sbin/nologin : người dùng không thể đăng nhập trên hệ thống )
						: user id   = 0 -> là người dùng admin root
			-passwd     //	set mật khẩu 						pass nguoidung
			-su         //	switch user : thay đổi người dùng   su nguoidung
			-userdel    // xóa người dùng 

			-chown   // change own thay đổi sở hữu 
						chown nguoidung file.txt 
						chown .nguoidung file.txt // thay đổi nhóm sở hữu
						chown :nguoidung file.txt // tương tự như trên 
						
# thay đổi quyền của file read(r) write(w) execute(x)
			chmod -command file.txt
			
				
# một file được sở hữu bởi 3 đối tượng
				-/ bằng chữ  
						User owner  -> u
						Group owner -> g
						Others      -> o
						rwx : read write  execute
					chmod u-rwx,g-rwx,o-rwx // bỏ hết quyên 
					chmod u+w file.txt ,  chmod g+rw file.txt  , chmod o+we file.txt	
					chmod o-wx file.txt   // loại bỏ quyền  
					
				-/ bẳng số  4(read) 2(write)  1(execute) 
				
				chmod 124 file.txt  // user : execute ; group write ; other read  
				chmod 776 file.txt  // user : full ; group full  ; other read + write 
				chmod 000 file.txt // xóa hết quyền
